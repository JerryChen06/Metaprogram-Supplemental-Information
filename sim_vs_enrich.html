data = transpose(raw_data)

// FIX: Find all cancer types that lack a predefined color (e.g., the Cell Lines)
missing_color_cancers = [...new Set(data.filter(d => !d.color).map(d => d.cancer_type))];

// Generate an automatic color palette for them using D3's built-in Tableau10 scheme
color_generator = d3.scaleOrdinal(d3.schemeTableau10).domain(missing_color_cancers);

// Map colors AND inject the generated colors back into the data objects
cancer_color_map = new Map(data.map(d => {
  if (!d.color) {
    d.color = color_generator(d.cancer_type); // Injects valid hex code so Plot "identity" works
  }
  return [d.cancer_type, d.color];
}))

global_x = d3.extent(data, d => d.gene_sim)
global_y = d3.extent(data, d => d.log2FC)

// 2. Main Toggles
viewof dataset_choice = Inputs.radio(["Human Cancer", "Cell Line"], {value: "Human Cancer", label: "Dataset:"})
viewof color_choice = Inputs.radio(["Cancer Type", "Hybrid Identity"], {value: "Cancer Type", label: "Colour By:"})

// 3. Dynamic Cancer Type Checkboxes
available_cancers = [...new Set(data.filter(d => d.dataset === dataset_choice).map(d => d.cancer_type))].sort()

// 4. Render Checkboxes with dynamic HTML colored squares AND layout overrides
viewof selected_cancers = {
  const chk = Inputs.checkbox(available_cancers, {
    value: available_cancers, 
    label: "Cancer Types:",
    format: x => {
      if (color_choice === "Cancer Type") {
        return html`<span style="color: ${cancer_color_map.get(x)}; font-size: 1.2em; vertical-align: middle;">■</span> ${x}`;
      } else {
        return x;
      }
    }
  });
  
  // Force single column layout
  const allLabels = chk.querySelectorAll("label");
  allLabels.forEach(label => {
    if (label.querySelector("input[type='checkbox']")) {
      label.style.display = "block";
      label.style.margin = "6px 0"; 
    }
  });

  // Prevent layout jumping by locking container height
  const firstCheckbox = chk.querySelector("input[type='checkbox']");
  if (firstCheckbox) {
    const container = firstCheckbox.closest("div");
    container.style.maxHeight = "55vh";      
    container.style.overflowY = "auto";      
    container.style.paddingRight = "10px";   
  }
  
  return chk;
}
filtered_data = data.filter(d => 
  d.dataset === dataset_choice && 
  selected_cancers.includes(d.cancer_type)
)

// 2. Render Plot
Plot.plot({
  height: 750,
  grid: true,
  
  // Apply the locked domains calculated from the raw dataset
  x: { label: "Gene Similarity", domain: global_x },
  y: { label: "log2FC", domain: global_y },
  
  // Color mapping logic
  color: color_choice === "Cancer Type" ? 
    { type: "identity" } : 
    { 
      legend: true, 
      domain: ["Activity-Activity", "Activity-Lineage", "Lineage-Lineage", "Other"], 
      range: ["#E41A1C", "#377EB8", "#4DAF4A", "#984EA3"] 
    },
    
  marks: [
    // Dashed red line at y = 1
    Plot.ruleY([1], {stroke: "red", strokeDasharray: "4", strokeWidth: 2}),
    
    // Scatterplot points
    Plot.dot(filtered_data, {
      x: "gene_sim",
      y: "log2FC",
      fill: color_choice === "Cancer Type" ? "color" : "hybrid_identity",
      r: 4,
      fillOpacity: 0.8,
      stroke: "black",
      strokeWidth: 0.5,
      title: "hover_text",  
      tip: true             
    })
  ]
})
